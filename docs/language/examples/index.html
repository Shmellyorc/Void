Threading + GC + lock integrationManaged workers, structured lock cleanup, and forced collection share one runtime model.
VOID
using Void;
using Void.Threading;

public sealed class Counter
{
    public int Value;
}

Counter counter = new Counter();

Thread first = new Thread(() =>
{
    lock (counter)
    {
        counter.Value++;
        GC.Collect();
    }
});

Thread second = new Thread(() =>
{
    lock (counter)
    {
        counter.Value++;
    }
});

first.Start();
second.Start();
first.Join();
second.Join();

Console.WriteLine(counter.Value == 2);
